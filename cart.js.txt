class Cart {
    constructor() {
        this.items = JSON.parse(localStorage.getItem('cp50_cart')) || [];
        this.updateUI();
    }
    addItem(product, qty = 1) {
        const existing = this.items.find(i => i.id === product.id);
        if (existing) existing.quantity += qty;
        else this.items.push({ ...product, quantity: qty });
        this.save(); this.updateUI();
    }
    removeItem(id) {
        this.items = this.items.filter(i => i.id !== id);
        this.save(); this.updateUI();
    }
    updateQty(id, qty) {
        const item = this.items.find(i => i.id === id);
        if (item) {
            item.quantity = qty;
            if (qty <= 0) this.removeItem(id);
            else { this.save(); this.updateUI(); }
        }
    }
    clear() { this.items = []; this.save(); this.updateUI(); }
    getTotal() { return this.items.reduce((sum, i) => sum + (i.price * i.quantity), 0); }
    getCount() { return this.items.reduce((sum, i) => sum + i.quantity, 0); }
    save() { localStorage.setItem('cp50_cart', JSON.stringify(this.items)); }
    updateUI() {
        const countEl = document.getElementById('cartCount');
        const itemsEl = document.getElementById('cartItems');
        const totalEl = document.getElementById('totalAmount');
        
        if (countEl) countEl.textContent = this.getCount();
        if (totalEl) totalEl.textContent = `$${this.getTotal().toFixed(2)}`;
        
        if (itemsEl) {
            if (!this.items.length) {
                itemsEl.innerHTML = '<p class="empty-cart">Tu carrito está vacío</p>';
            } else {
                itemsEl.innerHTML = this.items.map(i => `
                    <div class="cart-item">
                        <div class="cart-item-info">
                            <h4>${i.name}</h4>
                            <p>$${i.price.toFixed(2)} x ${i.quantity}</p>
                        </div>
                        <div class="cart-actions">
                            <div class="qty-controls">
                                <button onclick="cart.updateQty(${i.id}, ${i.quantity - 1})">-</button>
                                <span>${i.quantity}</span>
                                <button onclick="cart.updateQty(${i.id}, ${i.quantity + 1})">+</button>
                            </div>
                            <button class="btn-remove" onclick="cart.removeItem(${i.id})"><i class="fas fa-trash"></i></button>
                        </div>
                    </div>
                `).join('');
            }
        }
    }
}
const cart = new Cart();
