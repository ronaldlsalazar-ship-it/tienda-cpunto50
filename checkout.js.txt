document.addEventListener('DOMContentLoaded', () => {
    const checkoutBtn = document.getElementById('checkoutBtn');
    const closeCheckout = document.getElementById('closeCheckout');
    const checkoutModal = document.getElementById('checkoutModal');
    const checkoutForm = document.getElementById('checkoutForm');

    if (checkoutBtn) {
        checkoutBtn.addEventListener('click', () => {
            if (cart.items.length === 0) return alert('Tu carrito está vacío');
            document.getElementById('orderSummary').innerHTML = cart.items.map(i => 
                `<div class="summary-row"><span>${i.name} x${i.quantity}</span><span>$${(i.price * i.quantity).toFixed(2)}</span></div>`
            ).join('') + `<div class="summary-total"><span>Total:</span><span>$${cart.getTotal().toFixed(2)}</span></div>`;
            checkoutModal.classList.add('active');
        });
    }

    if (closeCheckout) closeCheckout.addEventListener('click', () => checkoutModal.classList.remove('active'));

    if (checkoutForm) {
        checkoutForm.addEventListener('submit', (e) => {
            e.preventDefault();
            const orderData = {
                cliente: document.getElementById('customerName').value,
                email: document.getElementById('customerEmail').value,
                telefono: document.getElementById('customerPhone').value,
                direccion: document.getElementById('customerAddress').value,
                banco: document.getElementById('bankSelect').value,
                items: cart.items,
                total: cart.getTotal(),
                fecha: new Date().toLocaleString()
            };
            
            // Aquí iría la integración con tu pasarela de pago o envío a WhatsApp
            console.log('Pedido generado:', orderData);
            alert(`¡Pedido recibido con éxito, ${orderData.cliente}!\nTotal: $${orderData.total.toFixed(2)}\n\nUn asesor se comunicará contigo para coordinar el pago y la entrega.`);
            cart.clear();
            checkoutModal.classList.remove('active');
            checkoutForm.reset();
        });
    }
});
