let products = [], services = [], config = {};

document.addEventListener('DOMContentLoaded', async () => {
    const data = await dataLoader.loadAll();
    products = data.products;
    services = data.services;
    config = data.config;

    renderProducts(products);
    renderServices();
    populateSelects();
    initFilters();
    initCartModal();
});

function renderProducts(list) {
    const grid = document.getElementById('productsGrid');
    if (!list.length) { grid.innerHTML = '<p class="no-results">No se encontraron productos.</p>'; return; }
    
    grid.innerHTML = list.map(p => `
        <div class="product-card">
            <div class="product-image"><i class="fas ${p.icon}"></i></div>
            <div class="product-info">
                <span class="product-category">${p.category.toUpperCase()}</span>
                <h3 class="product-name">${p.name}</h3>
                <p class="product-desc">${p.description}</p>
                <div class="product-price">$${p.price.toFixed(2)}</div>
                <span class="stock-badge ${p.stock > 0 ? 'in-stock' : 'made-to-order'}">
                    ${p.stock > 0 ? `En Stock (${p.stock})` : 'Bajo Pedido'}
                </span>
                <button class="btn-add-cart" onclick="addToCart(${p.id})">
                    <i class="fas fa-shopping-cart"></i> Agregar
                </button>
            </div>
        </div>
    `).join('');
}

function renderServices() {
    const grid = document.getElementById('servicesGrid');
    if (!grid) return;
    grid.innerHTML = services.map(s => `
        <div class="service-card">
            <i class="fas ${s.icon}"></i>
            <h3>${s.name}</h3>
            <p>${s.description}</p>
        </div>
    `).join('');
}

function populateSelects() {
    const bankSelect = document.getElementById('bankSelect');
    if (bankSelect && config.bancos) {
        bankSelect.innerHTML = '<option value="">Seleccionar banco...</option>' + 
            config.bancos.map(b => `<option value="${b}">${b}</option>`).join('');
    }
}

function addToCart(id) {
    const product = products.find(p => p.id === id);
    if (product) {
        cart.addItem(product);
        showNotification(`${product.name} agregado al carrito`);
    }
}

function initFilters() {
    const catFilter = document.getElementById('filterCategory');
    if (catFilter) {
        catFilter.addEventListener('change', (e) => {
            const filtered = e.target.value === 'all' ? products : products.filter(p => p.category === e.target.value);
            renderProducts(filtered);
        });
    }
}

function initCartModal() {
    const modal = document.getElementById('cartModal');
    document.getElementById('cartBtn').addEventListener('click', () => modal.classList.add('active'));
    document.getElementById('closeCart').addEventListener('click', () => modal.classList.remove('active'));
    modal.addEventListener('click', (e) => { if (e.target === modal) modal.classList.remove('active'); });
}

function showNotification(msg) {
    const notif = document.getElementById('notification');
    document.getElementById('notificationText').textContent = msg;
    notif.classList.add('show');
    setTimeout(() => notif.classList.remove('show'), 3000);
}

window.addToCart = addToCart;
